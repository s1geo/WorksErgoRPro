Examples

Example. Print angle between side2 and side3 in a triangle

from vcScript import *
from vcHelpers.Math import *
 
#angle in radians
print sssa(1.0, 1.0, 1.0)
 
#angle in degrees
print (sssa(1.0,1.0,1.0)*180.0)/PI

Example. Print expression of 4x4 matrix

from vcHelpers.Math import *
import vcMatrix
 
#prints Tx(100).Ty(100).Tz(100)
m = vcMatrix.new()
m.translateAbs(100,100,100)
print matrixExpr(m)
