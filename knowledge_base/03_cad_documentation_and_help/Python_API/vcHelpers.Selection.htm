Examples

Example. Print all point coordinates for all lines in the 3D world

from vcScript import *
from vcHelpers.Selection import *
 
for Is in filterGeosets(getAllComponentsGeosets(), VC_COMPACTLINESET):
  for p1,p2 in Is.Lines:
    print p1.X, p1.Y, p1.Z, p2.X, p2.Y, p2.Z
